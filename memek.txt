    redirect('login.php');

// ===== PROSES FORM =====
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    $action = $_POST['action'];
    $id = (int)($_POST['id'] ?? 0);
    $name = trim($_POST['name'] ?? '');
    $description = trim($_POST['description'] ?? '');
    
    if (empty($name)) {
        $_SESSION['error'] = 'Nama fasilitas harus diisi.';
    } else {
        // Upload file
        $image = null;
        if (!empty($_FILES['image']['name'])) {
            // Buat folder jika belum ada
            $upload_dir = '../uploads/facilities/';
            if (!file_exists($upload_dir)) {
                mkdir($upload_dir, 0777, true);
            }
            
            $upload = uploadFile($_FILES['image'], $upload_dir);
            if ($upload['success']) {
                $image = $upload['file'];
                // Simpan path relatif dari root (tanpa ../)
                $image = str_replace('../', '', $image);
            } else {
                $_SESSION['error'] = $upload['message'];
            }
        }
        
        if (!isset($_SESSION['error'])) {
            if ($action === 'edit' && $id > 0) {
                // Edit
                if ($image) {
                    // Hapus file lama
                    $stmt = $pdo->prepare("SELECT image FROM facilities WHERE id = ?");
                    $stmt->execute([$id]);
                    $old_image = $stmt->fetchColumn();
                    if ($old_image && file_exists('../' . $old_image)) {
                        unlink('../' . $old_image);
                    }
                    $stmt = $pdo->prepare("UPDATE facilities SET name = ?, description = ?, image = ? WHERE id = ?");
                    $stmt->execute([$name, $description, $image, $id]);
                } else {
                    $stmt = $pdo->prepare("UPDATE facilities SET name = ?, description = ? WHERE id = ?");
                    $stmt->execute([$name, $description, $id]);
                }
                $_SESSION['message'] = 'Fasilitas berhasil diupdate.';
            } elseif ($action === 'add') {
                // Tambah
                $stmt = $pdo->prepare("INSERT INTO facilities (name, description, image) VALUES (?, ?, ?)");
                $stmt->execute([$name, $description, $image]);
                $_SESSION['message'] = 'Fasilitas berhasil ditambahkan.';
            }
        }
    }
    
    // Redirect untuk menghindari double submit
    header("Location: " . $_SERVER['PHP_SELF']);
    exit();
}

// ===== HAPUS =====
if (isset($_GET['delete'])) {
    $id = (int)$_GET['delete'];
    
    // Ambil path file untuk dihapus
    $stmt = $pdo->prepare("SELECT image FROM facilities WHERE id = ?");
    $stmt->execute([$id]);
    $img = $stmt->fetchColumn();
    
    // Hapus file
    if ($img && file_exists('../' . $img)) {
        unlink('../' . $img);
    }
    
    $stmt = $pdo->prepare("DELETE FROM facilities WHERE id = ?");
    $stmt->execute([$id]);
    $_SESSION['message'] = 'Fasilitas berhasil dihapus.';
    redirect('facilities.php');
}

// ===== AMBIL PESAN =====

    $success = $_SESSION['message'];
    unset($_SESSION['message']);
}
if (isset($_SESSION['error'])) {

<script>
    function editFacility(id, name, description, image) {
        document.getElementById('modalFacilityTitle').innerHTML = '<i class="fas fa-edit"></i> Edit Fasilitas';
        document.getElementById('btnSubmitText').textContent = 'Update';
        document.getElementById('formAction').value = 'edit';
        
        document.getElementById('edit_id').value = id;
        document.getElementById('edit_name').value = name;
        document.getElementById('edit_description').value = description || '';
        
        const imgContainer = document.getElementById('currentImageContainer');
        const imgPreview = document.getElementById('currentImagePreview');
        if (image && image !== '') {
            imgPreview.src = '../' + image;
            imgContainer.style.display = 'block';
        } else {
            imgContainer.style.display = 'none';
        }
        
        document.getElementById('edit_image').value = '';
        
        var myModal = new bootstrap.Modal(document.getElementById('modalFacility'));
        myModal.show();
    }
    
    document.getElementById('modalFacility').addEventListener('hidden.bs.modal', function () {
        document.getElementById('modalFacilityTitle').innerHTML = '<i class="fas fa-plus-circle"></i> Tambah Fasilitas';
        document.getElementById('btnSubmitText').textContent = 'Simpan';
        document.getElementById('formAction').value = 'add';
        document.getElementById('edit_id').value = 0;
        document.getElementById('edit_name').value = '';
        document.getElementById('edit_description').value = '';
        document.getElementById('edit_image').value = '';
        document.getElementById('currentImageContainer').style.display = 'none';
    });
    
    document.getElementById('btnTambah').addEventListener('click', function() {
        document.getElementById('modalFacilityTitle').innerHTML = '<i class="fas fa-plus-circle"></i> Tambah Fasilitas';
        document.getElementById('btnSubmitText').textContent = 'Simpan';
        document.getElementById('formAction').value = 'add';
        document.getElementById('edit_id').value = 0;
        document.getElementById('edit_name').value = '';
        document.getElementById('edit_description').value = '';
        document.getElementById('edit_image').value = '';
        document.getElementById('currentImageContainer').style.display = 'none';
    });
</script>
